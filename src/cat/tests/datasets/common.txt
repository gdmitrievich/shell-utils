# 	VAR	IABLES
TARGET = s21_cat
BUILD_DIR = ../../build

CC = gcc
CFLAGS = -Wall -Wextra -Werror -std=c11

RM = rm -rf

# TOOLS
CLANG_FORMAT_OPTS = -i
CPPCHECK_OPTS = --verbose --enable=all --template=gcc --error-exitcode=1 --language=c

# SOURCES
SRCS = $(wildcard *.c)
OBJS = $(SRCS:%.c=$(BUILD_DIR)/%.o)

# TARGETS
all: ${TARGET}
.PHONY: all

rebuild: full_clean all
.PHONY: rebuild





run: all tools
	@./${TARGET}
.PHONY: run

${TARGET}: ${OBJS}
	${CC} ${CFLAGS} ${OBJS} -o ${TARGET}

${BUILD_DIR}/%.o : %.c
	@mkdir -p $(@D)
	$(CC) $(CFLAGS) -c $? -o $@

clean:
	${RM} ${OBJS}
.PHONY: clean

full_clean: clean
	${RM} ${TARGET}
.PHONY: full_clean

tools:
	clang-format -i ${SRCS}
	cppcheck ${CPPCHECK_OPTS} ${SRCS}
.PHONY: tools
